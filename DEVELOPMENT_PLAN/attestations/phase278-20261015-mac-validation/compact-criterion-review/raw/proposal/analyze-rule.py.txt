#!/usr/bin/env python3
"""Independent exact arithmetic examples; no model execution or gate claim."""
import hashlib
import json
import math
from fractions import Fraction
from pathlib import Path


def tail(n, classes, correct):
    return Fraction(
        sum(math.comb(n, k) * (classes - 1) ** (n - k) for k in range(correct, n + 1)),
        classes**n,
    )


tiny_previous = tail(1000, 200, 14)
tiny_critical = tail(1000, 200, 15)
assert tiny_previous > Fraction(1, 3000) >= tiny_critical
skewed_tail = tail(1000, 2, 601)
assert skewed_tail < Fraction(1, 3000)
# A 600/400-label example, always-majority except one uniformly chosen index:
# if that one index has minority label (probability .4), 601 are correct.
assert Fraction(400, 1000) == Fraction(2, 5)
assert 601 > 600

target = [Fraction(-1), Fraction(1)]
mean = sum(target) / len(target)
sst = sum((y - mean) ** 2 for y in target)
constant_sse = sum((y - 0) ** 2 for y in target)
tiny_predictions = [Fraction.from_float(-1e-16), Fraction.from_float(1e-16)]
tiny_sse = sum((y - p) ** 2 for y, p in zip(target, tiny_predictions))
assert constant_sse == sst and tiny_sse < sst

inputs = [

    "src/JitML/Checkpoint/WeightCodec.hs",
]
result = {

    "tiny_n1000_classes200": {

        "tail_at_15": float(tiny_critical),
